using FileManager.Infrastructure.Models;
using FileManager.Services;
using Microsoft.AspNetCore.Mvc;

namespace FileManager.WebApi.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    public class TextFilesController : ControllerBase
    {
        private readonly ICrudServiceAsync<TextFile> _service;

        public TextFilesController(ICrudServiceAsync<TextFile> service)
        {
            _service = service;
        }

        [HttpGet]
        public async Task<IActionResult> GetAll()
        {
            var files = await _service.ReadAllAsync();
            return Ok(files);
        }

        [HttpGet("{id}")]
        public async Task<IActionResult> Get(Guid id)
        {
            var file = await _service.ReadAsync(id);
            if (file == null) return NotFound();
            return Ok(file);
        }

        [HttpPost]
        public async Task<IActionResult> Create(TextFile file)
        {
            await _service.CreateAsync(file);
            await _service.SaveAsync();
            return CreatedAtAction(nameof(Get), new { id = file.Id }, file);
        }

        [HttpPut("{id}")]
        public async Task<IActionResult> Update(Guid id, TextFile updated)
        {
            var file = await _service.ReadAsync(id);
            if (file == null) return NotFound();

            file.FileName = updated.FileName;
            file.Extension = updated.Extension;
            file.SizeBytes = updated.SizeBytes;
            file.FolderId = updated.FolderId;

            await _service.UpdateAsync(file);
            await _service.SaveAsync();

            return Ok(file);
        }

        [HttpDelete("{id}")]
        public async Task<IActionResult> Delete(Guid id)
        {
            var file = await _service.ReadAsync(id);
            if (file == null) return NotFound();

            await _service.RemoveAsync(file);
            await _service.SaveAsync();
            return NoContent();
        }
    }
}
